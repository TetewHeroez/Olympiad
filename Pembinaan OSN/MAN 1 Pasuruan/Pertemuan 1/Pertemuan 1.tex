\section{Aturan Penjumlahan dan Perkalian}

\begin{frame}{\insertsection}{\insertsection}
  \begin{definisi}[Aturan Penjumlahan]
    Jika ada sebanyak $m$ pilihan pada kejadian pertama dan ada sebanyak $n$ pilihan pada kejadian kedua, dan kedua kejadian itu tidak dapat dilakukan dalam waktu yang sama, maka ada $m+n$ cara untuk memilih satu dari kejadian tersebut. Biasanya pada soal terdapat kata \textbf{"atau"}.
  \end{definisi}
\end{frame}

\begin{frame}{\insertsection}{\insertsection}
  \begin{contoh}
    Budi akan menghadiri pesta. Ia harus memilih satu celana dari 4 celana merahnya dan 3 celana birunya. Berapa banyak cara Budi memilih celana?

    \textbf{Penyelesaian : }\\
    Karena Budi tidak mungkin menggunakan dua celana secara bersamaan, sehingga ia harus memilih $1$ celana dari $4$ celana merah \textbf{atau} $3$ celana biru. Dengan aturan perkalian, banyaknya cara Budi memilih celana adalah
    $$4+3=7 \text{ cara}.$$
  \end{contoh}
\end{frame}

\begin{frame}{\insertsection}{\insertsection}
  \begin{definisi}[Aturan Perkalian]
    Aturan perkalian sering kali disebut dengan \textit{filling slot }aturan pengisian tempat.
    Misalkan ada $n$ tempat yang tersedia, dengan tempat ke-$1$ memiliki cara sebanyak $k_1$, tempat ke-$2$ memiliki cara sebanyak $k_2$, dan seterusnya sampai tempat ke-$n$ memiliki cara sebanyak $k_n$. Dengan demikian, banyaknya cara mengisi tempat adalah
    \begin{align}
      \uline{k_1} \ \uline{k_2} \ \uline{k_3} \ ... \ \uline{k_n} = k_1 \times k_2 \times k_3 \times ... \times k_n.
    \end{align}
  \end{definisi}
\end{frame}

\begin{frame}{\insertsection}{\insertsection}
  \begin{contoh}
    Misalkan terdapat $2$ buah baju dan $3$ buah celana. Berapa banyak seseorang dapat memilih baju dan celana yang akan Ia pakai?

    \textbf{Penyelesaian : }\\
    Misalkan tempat ke-$1$ adalah banyak cara memilih baju dan tempat ke-$2$ adalah banyak cara memilih celana. Karena banyaknya baju ada $2$ dan celana ada $3$, akibatnya
    \begin{align}
      \uline{2} \ \uline{3}=2 \times 3 = 6 \notag
    \end{align}
  \end{contoh}
\end{frame}

\section{Permutasi}
\subsection{Permutasi Unsur Berbeda}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Permutasi $r$ objek yang diambil dari $n$ objek berbeda adalah $P_r^n$ yang didefinisikan dengan
    \begin{align}
      P_r^n = \frac{n!}{(n-r)!}
    \end{align}
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Berapa banyak cara menyusun 3 huruf berbeda dari 4 huruf K, L, M, N ?

    \textbf{Penyelesaian : }\\
    Jika menggunakan \textit{filling slot} yang sebelumnya kita ketahui, maka banyak caranya adalah
    \begin{align}
      \uline{4} \ \uline{3} \  \uline{2} = 4 \times 3 \times 2 = 24 \notag
    \end{align}
    Atau, karena kita mengambil 3 dari 4 objek berbeda, akibatnya kita dapat menggunakan permutasi dari unsur-unsur yang berbeda, dengan $r=3$ dan $n=4$.
    \begin{align}
      P_3^4 = \frac{4!}{(4-3)!}=\frac{4!}{1!}=\frac{4 \times 3 \times 2 \times 1}{1}=24. \notag
    \end{align}
  \end{contoh}
\end{frame}

\subsection{Permutasi Unsur Sama}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Banyak permutasi $n$ unsur yang memuat $k_1$ unsur yang sama, $k_2$ unsur yang sama, dan seterusnya sampai dengan $k_i$ unsur yang sama, dengan $k_1+k_2+k_3+...+k_i=n$ ditentukan dengan rumus
    \begin{align}
      P=\frac{n!}{k_1!k_2!...k_i!}
    \end{align}
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Ada berapa banyak susunan yang dapat dibentuk dari huruf-huruf B,E,B,E,K?

    \textbf{Penyelesaian : }\\
    Banyaknya huruf B ada 2, E ada 2, dan K ada 1. Dengan demikian, banyak susunan yang dapat dibuat
    \begin{align}
      P=\frac{5!}{2!2!1!}=\frac{5 \times 4 \times 3 \times 2 \times 1}{2 \times 1 \times 2 \times 1}=30. \notag
    \end{align}
  \end{contoh}
\end{frame}

\subsection{Permutasi Siklis}
\begin{frame}{\insertsection}{\insertsubsection}
  Bagaimana jika ada $n$ orang duduk melingkari meja bundar? Ada berapa banyak cara menyusunnya?
  \begin{teorema}
    Banyaknya permutasi siklis dari $n$ unsur tersebut dirumuskan dengan
    \begin{align}
      P(\text{siklis}) = (n-1)!
    \end{align}
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Jika terdapat tiga orang yang duduk pada tiga kursi yang membentuk suatu lingkaran, maka ada berapa banyak susunan yang mungkin terjadi?

    \textbf{Penyelesaian : }\\
    Permutasi siklis dengan $n=3$ didapat
    \begin{align}
      P(\text{siklis})=(3-1)!=2!=2. \notag
    \end{align}
  \end{contoh}
\end{frame}

\section{Kombinasi}
\begin{frame}{\insertsection}
  \begin{definisi}
    Suatu kombinasi $r$ unsur yang diambil dari $n$ unsur berbeda yang tersedia adalah suatu pilihan dari r unsur tadi\textbf{ tanpa memperhatikan urutannya}. Banyaknya kombinasi $r$ unsur yang diambil dari $n$ unsur yang tersedia dengan $r \leq n$ dirumuskan dengan
    \begin{align}
      C_r^n=\frac{n!}{(n-r)!r!}
    \end{align}
  \end{definisi}
\end{frame}

\begin{frame}{\insertsection}
  \begin{contoh}
    Berapa banyak cara memilih 2 perwakilan lomba dari 5 siswa yang ada?

    \textbf{Penyelesaian : }\\
    \begin{align}
      C_2^5=\frac{5!}{(5-2)!2!}=\frac{5!}{3!2!}=10. \notag
    \end{align}
  \end{contoh}
\end{frame}


\section{Distribusi dan Partisi}
\begin{frame}{\insertsection}
  \begin{definisi}
    Sebuah partisi dari A adalah kumpulan dari himpunan bagian tak kosong yang saling lepas (disjoint subsets) dari A yang gabungannya adalah seluruh A.
  \end{definisi}

  Selanjutnya, partisi $A$ elemen ke dalam $k$ bagian dapat diartikan menjadi distribusi $A$ elemen ke dalam $k$ kotak.
  Untuk menyelsaikan soal tipe ini, dapat menggunakan aturan penjumlahan dan perkalian, permutasi, kombinasi, dan \textbf{stars and bars}.
\end{frame}

\begin{frame}{\insertsection}
  \begin{teorema}[Stars and Bars]
    Misalkan $r$ dan $k$ adalah bilangan bulat non-negatif. Banyaknya solusi non-negatif dari
    $$x_1+x_2+\ldots+x_k=r$$
    adalah $\begin{pmatrix}
        r+k-1 \\k-1
      \end{pmatrix}=\begin{pmatrix}
        r+k-1 \\r
      \end{pmatrix}$.
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Berapa solusi bilangan bulat dari
    $$w_1+w_2+w_3+w_4=7,$$
    dengan $w_1 \geq 1$ dan $w_3 \geq 2$.\\
    \textbf{Penyelesaian:}\\
    Jika menggunnakan teorema stars and bars, maka diperoleh nilai $w_1,w_2,w_3,w_4 \geq 0.$ Namun dengan batasan pada soal, dimisalkan $w_1=y_1+1, w_2=y_2, w_3=y_3+2,$ dan $w_4=y_4$ dengan $y_1,y_2,y_3,y_4 \geq 0.$ Diperoleh
    $$y_1+y_2+y_3+y_4=4,$$
    dengan stars and bar diperoleh banyaknya solusi
    $\begin{pmatrix}
        4+4-1 \\4
      \end{pmatrix}=\begin{pmatrix}
        7 \\ 4
      \end{pmatrix}=\frac{7.6.5}{3.2.1}=35.$
  \end{contoh}
\end{frame}

\section{Prinsip Injeksi dan Bijeksi}
\begin{frame}{\insertsection}
  \begin{definisi}
    Misalkan $A$ dan $B$ adalah himpunan. Suatu fungsi $f : A \to B$ disebut
    \textit{injektif} (atau satu-satu) jika untuk setiap $a_1, a_2 \in A$ berlaku
    \[
      f(a_1) = f(a_2) \Rightarrow a_1 = a_2.
    \]
  \end{definisi}

  \begin{teorema}
    Misalkan $A$ dan $B$ adalah himpunan hingga. Jika terdapat fungsi injektif
    $f : A \to B$, maka berlaku
    \[
      |A| \leq |B|.
    \]
  \end{teorema}
\end{frame}


\begin{frame}{\insertsection}
  \begin{definisi}\textbf{(Fungsi Surjektif)}
    Misalkan $f : A \to B$ adalah fungsi. Fungsi $f$ disebut
    \textit{surjektif} jika untuk setiap $b \in B$, terdapat $a \in A$ sehingga
    \[
      f(a) = b.
    \]
  \end{definisi}

  \begin{definisi}
    Misalkan $A$ dan $B$ adalah himpunan. Suatu fungsi $f : A \to B$ disebut
    \textit{bijektif} jika $f$ bersifat injektif dan surjektif.
  \end{definisi}

  \begin{teorema}
    Misalkan $A$ dan $B$ adalah himpunan hingga. Jika terdapat fungsi bijektif
    $f : A \to B$, maka berlaku
    \[
      |A| = |B|.
    \]
  \end{teorema}
\end{frame}
