\section{Ketaksamaan}
\begin{frame}{\insertsection}
  \textbf{Ketaksamaan} adalah suatu pernyataan yang menyatakan bahwa suatu nilai lebih besar, lebih kecil, atau sama dengan nilai lainnya. Biasanya disimbolkan dengan tanda $>$, $<$, $\geq$, atau $\leq$.
  \begin{block}{Kegunaan menganalisa ketaksamaan aljabar}
    \begin{itemize}
      \item \textbf{Optimasi (Maksimum/Minimum)}: Menentukan nilai maksimum atau minimum dari suatu fungsi atau ekspresi.
      \item \textbf{Pembuktian}: Membuktikan bahwa suatu ekspresi aljabar selalu lebih besar dari ekspresi lainnya untuk kondisi apa pun.
      \item \textbf{Batas Solusi (Bounding)}: Di soal Teori Bilangan atau Kombinatorika, ketaksamaan sering dipakai untuk membatasi rentang pencarian agar kita tidak perlu mencoba angka satu per satu sampai tak terhingga.
    \end{itemize}
  \end{block}
\end{frame}
\begin{frame}{\insertsection}
  \begin{proposisi}[Ketaksamaan Trivial\label{prop:trivial}]
    Untuk setiap bilangan real $x$, berlaku $x^2 \geq 0$.
  \end{proposisi}
  \begin{contoh}
    Buktikan bahwa untuk setiap bilangan real non-negatif $x$ dan $y$, berlaku\label{cth:trivial}
    \[
      x+y \geq 2\sqrt{xy}.
    \]
    \onslide<2->{\textbf{Jawab}: Menggunakan \myref{prop:trivial}, kita dapat menulis
      \[
        (\sqrt{x} - \sqrt{y})^2 \geq 0 \implies x + y - 2\sqrt{xy} \geq 0 \implies x+y \geq 2\sqrt{xy}.
      \]
      Kesamaan terjadi jika dan hanya jika $x = y$.}
  \end{contoh}
  \onslide<2->{\myref{prop:trivial} adalah sebuah ide dasar untuk membuktikan banyak ketaksamaan lainnya, dalam hal ini adalah contoh ketaksamaan AM-GM yang akan kita bahas nanti.  }
\end{frame}
\subsection{Ketaksamaan Segitiga}
\begin{frame}{\insertsection}{\insertsubsection}
  Tinjau sebuah segitiga sembarang dengan titik $A$, $B$, dan $C$. Gunakan hukum cosinus dan fakta bahwa $-1\leq\cos(\theta) \leq 1$ untuk setiap garis yang menghubungkan titik-titik tersebut, maka dapat diperoleh:
  \begin{columns}[T]
    \begin{column}{0.65\textwidth}
      \begin{align*}
        \overline{AB}^2 & \leq \overline{AC}^2 + \overline{BC}^2 + 2 \cdot \overline{AC} \cdot \overline{BC} = (\overline{AC} + \overline{BC})^2, \\
        \overline{AC}^2 & \leq \overline{AB}^2 + \overline{BC}^2 + 2 \cdot \overline{AB} \cdot \overline{BC} = (\overline{AB} + \overline{BC})^2, \\
        \overline{BC}^2 & \leq \overline{AB}^2 + \overline{AC}^2 + 2 \cdot \overline{AB} \cdot \overline{AC} = (\overline{AB} + \overline{AC})^2.
      \end{align*}
      Sehingga diperoleh ketaksamaan segitiga untuk setiap sisi segitiga $ABC$ adalah
      \[
        \overline{AB} \leq \overline{AC} + \overline{BC}, \quad \overline{AC} \leq \overline{AB} + \overline{BC}, \quad \overline{BC} \leq \overline{AB} + \overline{AC}.
      \]
    \end{column}
    \begin{column}{0.3\textwidth}
      \noindent
      \begin{tikzpicture}[scale=0.9, every node/.style={font=\small}, >=stealth, rotate=90]
        \coordinate (B) at (0,0);
        \coordinate (C) at (4,0);
        \coordinate (A) at (5, 2.5);
        \draw[thick] (B) -- node[sloped,rotate=90] {$|||$} (A) -- node[sloped,rotate=90] {$|$} (C) -- node[sloped,rotate=90] {$||$} cycle;

        \fill[black] (A) circle (1.2pt) node[above=2pt] {$A$};
        \fill[black] (B) circle (1.2pt) node[left=2pt] {$B$};
        \fill[black] (C) circle (1.2pt) node[right=2pt] {$C$};
      \end{tikzpicture}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}[AMC 10B 2006]
    Dalam sebuah segitiga dengan panjang sisi bilangan bulat, panjang sisi pertama adalah tiga kali panjang sisi kedua, dan panjang sisi ketiga adalah $15$. Berapakah keliling maksimum yang mungkin dari segitiga tersebut?\\
    \onslide<2->{\textbf{Solusi:} Misalkan panjang sisi kedua adalah $x$, maka panjang sisi pertama adalah $3x$. Karena panjang sisi ketiga adalah $15$, kita dapat menggunakan ketaksamaan segitiga untuk menentukan nilai $x$:
      \begin{align*}
        3x + x > 15  \implies x > \frac{15}{4} = 3.75\quad\text{dan}\quad x + 15 > 3x \implies x < 7.5.
      \end{align*}
      Karena $x$ harus berupa bilangan bulat, maka nilai $x$ yang memenuhi adalah $4$, $5$, $6$, atau $7$. Sehingga keliling maksimum adalah
      $
        3x + x + 15 = 3(7) + 7 + 15 = \boxed{43}.
      $
    }
  \end{contoh}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  Dari sebuah segitiga, dapat diperoleh beberapa abstraksi ketaksamaan yang berlaku untuk bilangan real.
  \begin{teorema}[Ketaksamaan Segitiga]
    Untuk setiap bilangan real $a$ dan $b$, berlaku\label{thm:segitiga}
    \[
      |a + b| \leq |a| + |b|.
    \]
  \end{teorema}
  \begin{akibat}
    Untuk setiap bilangan real $a_1, a_2, \ldots, a_n$, berlaku\label{thm:segitiga_general}
    \[
      |a_1 + a_2 + \cdots + a_n| \leq |a_1| + |a_2| + \cdots + |a_n|.
    \]
  \end{akibat}
  Kesamaan pada \myref{thm:segitiga} dan \myref{thm:segitiga_general} terjadi \textbf{jika dan hanya jika} semua bilangan real-nya memiliki tanda yang sama (semua positif atau semua negatif).
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Tentukan semua bilangan real $x$ yang memenuhi persamaan berikut:\label{cth:triangle_inequality}
    $$|x^2 - 4x + 3| + |x^2 - 9| = |2x^2 - 4x - 6|$$

    \onslide<2->{\textbf{Solusi:}
      Misalkan ekspresi di dalam nilai mutlak pada ruas kiri sebagai
      \begin{align*}
        a  = x^2 - 4x + 3\quad\text{dan} \quad
        b  = x^2 - 9
      \end{align*}
      Perhatikan hasil penjumlahan dari $a$ dan $b$
      \begin{align*}
        a + b & = (x^2 - 4x + 3) + (x^2 - 9) = 2x^2 - 4x - 6
      \end{align*}}
  \end{contoh}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{exampleblock}{\myref{cth:triangle_inequality}}
    Bentuk $a+b$ tepat sama dengan ekspresi di dalam nilai mutlak pada ruas kanan persamaan. Sehingga, persamaan tersebut dapat ditulis kembali menjadi
    $$|a| + |b| = |a + b|$$
    Berdasarkan \myref{thm:segitiga}, kondisi kesamaan terjadi jika dan hanya jika $a$ dan $b$ tidak memiliki tanda yang berlawanan, yang berarti
    $$a \cdot b \ge 0\implies(x^2 - 4x + 3)(x^2 - 9) \ge 0\implies(x - 1)(x + 3)(x - 3)^2 \ge 0$$
    Jadi, himpunan penyelesaian dari persamaan tersebut adalah
    $$\{x \in \mathbb{R} \mid x \le -3 \text{ atau } x \ge 1\}$$
  \end{exampleblock}
\end{frame}
\subsection{Ketaksamaan QM-AM-GM-HM}
\begin{frame}{\insertsection}{\insertsubsection}
  Pertama kita definisikan beberapa jenis ``rataan'' yang akan digunakan.
  \begin{definisi}
    Misalkan $x_1, x_2, \ldots, x_n$ adalah bilangan real positif. Maka kita definisikan beberapa jenis rataan sebagai berikut:
    \begin{itemize}
      \item \textbf{Rataan Kuadrat} (\textit{Quadratic Mean}): $\displaystyle\boxed{\mathbf{QM} = \sqrt{\frac{x_1^2 + x_2^2 + \cdots + x_n^2}{n}}}$
      \item \textbf{Rataan Aritmetika} (\textit{Arithmetic Mean}): $\displaystyle\boxed{\mathbf{AM} = \frac{x_1 + x_2 + \cdots + x_n}{n}}$
      \item \textbf{Rataan Geometrik} (\textit{Geometric Mean}): $\displaystyle\boxed{\mathbf{GM} = \sqrt[n]{x_1 x_2 \cdots x_n}}$
      \item \textbf{Rataan Harmonik} (\textit{Harmonic Mean}): $\displaystyle\boxed{\mathbf{HM} = \frac{n}{\frac{1}{x_1} + \frac{1}{x_2} + \cdots + \frac{1}{x_n}}}$
    \end{itemize}
  \end{definisi}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Ketaksamaan QM-AM-GM-HM\label{thm:AM-GM}]
    Untuk setiap bilangan real positif $x_1, x_2, \ldots, x_n$, berlaku
    \[
      \mathbf{QM} \geq \mathbf{AM} \geq \mathbf{GM} \geq \mathbf{HM}.
    \]
    Kesamaan terjadi jika dan hanya jika $x_1 = x_2 = \cdots = x_n$.
  \end{teorema}
  Pada soal nantinya kita hanya perlu menerapkan salah satu dari ketaksamaan di atas (AM-GM, AM-HM, QM-AM, dll)
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Tentukan nilai minimum dari fungsi $f(x) = x + \dfrac{1}{x^2}$ untuk $x > 0$.\label{cth:AM-GM}

    \onslide<2->{\textbf{Solusi:}
      Jika kita langsung menerapkan Ketaksamaan AM-GM pada dua suku $x$ dan $\frac{1}{x^2}$, kita akan mendapatkan:
      $$f(x) = x + \frac{1}{x^2} \ge 2\sqrt{x \cdot \frac{1}{x^2}} = 2\sqrt{\frac{1}{x}}$$
      Perhatikan bahwa variabel $x$ tidak saling menghilangkan di ruas kanan. Ini berarti kita tidak mendapatkan batasan nilai berupa angka konstan (tetap).

      Untuk mengakalinya, kita harus menyamakan pangkat $x$ di pembilang dan penyebut saat dikalikan nanti. Karena di penyebut ada $x^2$, kita butuh $x^2$ di pembilang. Caranya adalah dengan memecah suku $x$ menjadi dua bagian yang sama, yaitu $\frac{x}{2} + \frac{x}{2}$.}
  \end{contoh}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{exampleblock}{\myref{cth:AM-GM}}
    Sekarang, terapkan Ketaksamaan AM-GM (\myref{thm:AM-GM}) untuk tiga suku
    \begin{align*}
      f(x) = \frac{x}{2} + \frac{x}{2} + \frac{1}{x^2} & \ge 3\sqrt[3]{\frac{\cancel{ x }}{2} \cdot \frac{\cancel{ x }}{2} \cdot \frac{1}{\cancel{ x^2 }}}= 3\sqrt[3]{\frac{1}{4}} = \frac{3}{2}\sqrt[3]{2}
    \end{align*}
    Kondisi kesamaan terjadi jika dan hanya jika $\dfrac{x}{2} = \dfrac{1}{x^2}$, yang berarti $x = \sqrt[3]{2}$. Jadi, nilai minimum dari $f(x)$ adalah $\dfrac{3}{2}\sqrt[3]{2}$, yang dicapai saat $x = \sqrt[3]{2}$.
  \end{exampleblock}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[AM-GM Berbobot (\textit{Weighted AM-GM})\label{thm:weighted-AM-GM}]
    Untuk setiap bilangan real positif $x_1, x_2, \ldots, x_n$ dan bobot $w_1, w_2, \ldots, w_n$ yang memenuhi $w_i > 0$ dan $\sum_{i=1}^n w_i = w$, berlaku
    \[
      \frac{w_1 x_1 + w_2 x_2 + \cdots + w_n x_n}{w} \geq \sqrt[w]{x_1^{w_1} x_2^{w_2} \cdots x_n^{w_n}}.
    \]
    Kesamaan terjadi jika dan hanya jika $x_i$ memiliki nilai yang sama untuk setiap $i=1,2,\dots,n$.
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Ketaksamaan Cauchy-Schwarz\label{thm:cauchy-schwarz}]
    Untuk setiap bilangan real $a_1, a_2, \ldots, a_n$ dan $b_1, b_2, \ldots, b_n$, berlaku
    \[
      (a_1^2 + a_2^2 + \cdots + a_n^2)(b_1^2 + b_2^2 + \cdots + b_n^2) \geq (a_1 b_1 + a_2 b_2 + \cdots + a_n b_n)^2.
    \]
    Kesamaan terjadi jika dan hanya jika terdapat suatu bilangan real $k$ sehingga $a_i = k b_i$ untuk setiap $i = 1, 2, \ldots, n$.
  \end{teorema}
  Ketaksamaan Cauchy-Schwarz dapat digeneralisasi untuk sebuah ruang vektor dengan hasil kali dalam, yaitu untuk setiap vektor $\mathbf{u}$ dan $\mathbf{v}$, berlaku
  \[
    |\langle \mathbf{u}, \mathbf{v} \rangle| \leq \|\mathbf{u}\| \cdot \|\mathbf{v}\|.
  \]
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{lema}[Lema Titu\label{lem:titu}]
    Untuk setiap bilangan real positif $a_1, a_2, \ldots, a_n$ dan $b_1, b_2, \ldots, b_n$, berlaku
    \[
      \frac{a_1^2}{b_1} + \frac{a_2^2}{b_2} + \cdots + \frac{a_n^2}{b_n} \geq \frac{(a_1 + a_2 + \cdots + a_n)^2}{b_1 + b_2 + \cdots + b_n}.
    \]
    Kesamaan terjadi jika dan hanya jika $\displaystyle\frac{a_1}{b_1} = \frac{a_2}{b_2} = \cdots = \frac{a_n}{b_n}$.
  \end{lema}
  \myref{lem:titu} diperoleh dengan menerapkan \myref{thm:cauchy-schwarz} pada dua barisan yaitu $a_i/\sqrt{b_i}$ dan $\sqrt{b_i}$.
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Diberikan bilangan real positif $x$ dan $y$ yang memenuhi $x + y = 2$. Tentukan nilai minimum
    \[
      \frac{x^4}{y} + \frac{y^4}{x} + \frac{2}{xy}\label{cth:CS}
    \]

    \onslide<2->{\textbf{Jawab:}

      Tinjau sebagai penjumlahan dua bagian $E_1 = \frac{x^4}{y} + \frac{y^4}{x}$ dan $E_2 = \frac{2}{xy}$. Terapkan Lema Titu (\myref{lem:titu})  pada $E_1$:
      \[
        E_1 = \frac{(x^2)^2}{y} + \frac{(y^2)^2}{x} \ge \frac{(x^2+y^2)^2}{x+y} = \frac{(x^2+y^2)^2}{2}
      \]}
  \end{contoh}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{exampleblock}{\myref{cth:CS}}
    Terapkan Ketaksamaan QM-AM (\myref{thm:AM-GM}) untuk menemukan batas dari $(x^2+y^2)$:
    \[
      \sqrt{\frac{x^2+y^2}{2}} \ge \frac{x+y}{2}=\frac{2}{2} = 1\implies x^2+y^2 \ge 2
    \]
    Substitusikan batas ini ke hasil evaluasi $E_1$:
    \[
      E_1 \ge \frac{(2)^2}{2} = 2
    \]
    Terapkan Ketaksamaan AM-GM (\myref{thm:AM-GM}) pada bagian penyebut $E_2$:
    \[
      \frac{x+y}{2} \ge \sqrt{xy} \implies\frac{2}{2} \ge \sqrt{xy} \implies 1 \ge \sqrt{xy} \implies xy \le 1
    \]
  \end{exampleblock}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{exampleblock}{\myref{cth:CS}}
    Karena $xy \le 1$ dan variabel bernilai positif, maka $\frac{1}{xy} \ge 1$. Sehingga
    \[
      E_2 = \frac{2}{xy} \ge \frac{2}{1} = 2
    \]
    Jumlahkan batas minimum dari masing-masing bagian untuk mendapatkan batas total:
    \[
      E_1 + E_2= \frac{x^4}{y} + \frac{y^4}{x} + \frac{2}{xy} \ge 2 + 2 = 4
    \]
    Kesamaan terpenuhi jika dan hanya jika $x = y = 1$. Jadi, nilai minimum dari ekspresi tersebut adalah $4$.
  \end{exampleblock}
\end{frame}

\subsection{Ketaksamaan Renata}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{catatan}
    Dalam literatur matematika, ketaksamaan renata seharusnya dikenal sebagai \textit{Rearrangement Inequality}. Jadi secara informal,``renata'' dapat diartikan ``penataan ulang''.
  \end{catatan}
  \begin{definisi}
    Permutasi $\sigma$ adalah suatu fungsi bijektif dari himpunan $\{1, 2, \ldots, n\}$ ke dirinya sendiri. Dengan kata lain, $\sigma$ adalah suatu cara untuk mengatur ulang elemen-elemen dalam himpunan tersebut.
  \end{definisi}
  \begin{contoh}
    Misalkan kita memiliki himpunan $\{1, 2, 3\}$. Beberapa contoh permutasi adalah:
    \begin{itemize}
      \item $\sigma_1$: $1 \mapsto 1$, $2 \mapsto 2$, $3 \mapsto 3$ (identitas)
      \item $\sigma_2$: $1 \mapsto 2$, $2 \mapsto 3$, $3 \mapsto 1$ (rotasi)
      \item $\sigma_3$: $1 \mapsto 1$, $2 \mapsto 3$, $3 \mapsto 2$ (transposisi)
    \end{itemize}
  \end{contoh}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Ketaksamaan Renata\label{thm:renata}]
    Untuk setiap bilangan real $a_1 \leq a_2 \leq \cdots \leq a_n$ dan $b_1 \leq b_2 \leq \cdots \leq b_n$, berlaku
    \[
      \sum_{i=1}^n a_i b_i \geq \sum_{i=1}^n a_i b_{\sigma(i)} \geq \sum_{i=1}^n a_i b_{n-i+1}
    \]
    untuk setiap permutasi $\sigma$ dari $\{1, 2, \ldots, n\}$.
  \end{teorema}
  Intuisi dari ketaksamaan ini adalah bahwa untuk memaksimalkan jumlah hasil kali antara dua barisan yang sudah diurutkan, kita harus mengalikan elemen-elemen yang berurutan (yaitu $a_i$ dengan $b_i$). Sebaliknya, untuk meminimalkan jumlah tersebut, kita harus mengalikan elemen-elemen yang berlawanan urutan (yaitu $a_i$ dengan $b_{n-i+1}$).
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{akibat}[Ketaksamaan Chebyshev]
    Untuk setiap bilangan real $a_1 \leq a_2 \leq \cdots \leq a_n$ dan $b_1 \leq b_2 \leq \cdots \leq b_n$, berlaku
    \[n\sum_{i=1}^n a_i b_i \geq \left(\sum_{i=1}^n a_i\right)\left(\sum_{i=1}^n b_i\right)\]
  \end{akibat}
  Ketaksamaan Chebyshev diperoleh dengan menjumlahkan ketaksamaan renata dengan mengambil permutasi-permutasi siklik pada $b_i$.
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Tentukan nilai minimum dari\label{cth:renata}
    \[
      \frac{x^3}{y} + \frac{y^3}{z} + \frac{z^3}{x}
    \]
    jika $x, y,$ dan $z$ adalah bilangan real positif yang memenuhi $x^2 + y^2 + z^2 = 12$.

    \onslide<2->{\textbf{Jawab:}
      Misalkan terdapat himpunan yang terdiri dari bilangan $x, y,$ dan $z$. Kita dapat mengurutkan elemen-elemen tersebut dari yang terbesar hingga terkecil. Sebut urutan ini sebagai $p \ge q \ge r$

      Dari urutan tersebut dapat diperoleh dua urutan yang lain yaitu
      \[ \frac{1}{p} \le \frac{1}{q} \le \frac{1}{r} \quad\text{dan}\quad p^3 \ge q^3 \ge r^3\]}
  \end{contoh}
\end{frame}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{exampleblock}{\myref{cth:renata}}
    Menurut ketaksamaan renata (\myref{thm:renata}), Nilai minimumnya terjadi ketika
    \[ p^3\left(\frac{1}{p}\right) + q^3\left(\frac{1}{q}\right) + r^3\left(\frac{1}{r}\right) = p^2 + q^2 + r^2 \]
    Mengingat $p, q, r$ hanyalah susunan ulang dari $x, y, z$, maka secara otomatis
    \[p^2 + q^2 + r^2 = x^2 + y^2 + z^2 =12.\]
    Batas bawah ini valid dan kesamaan terjadi apabila $x = y = z$. Mengingat persyaratannya adalah $3x^2 = 12 \implies x^2 = 4$, maka nilai minimum tersebut secara mutlak dicapai pada titik koordinat $(2, 2, 2)$. Jadi, nilai minimum dari ekspresi tersebut adalah 12.
  \end{exampleblock}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Ketaksamaan Jensen\label{thm:jensen}]
    Jika $f$ adalah fungsi konveks dan $x_1, x_2, \ldots, x_n$ adalah bilangan real dalam domain $f$, maka untuk setiap himpunan bobot $w_1, w_2, \ldots, w_n$ yang memenuhi $w_i \geq 0$ dan $\sum_{i=1}^n w_i = 1$, berlaku
    \[
      f\left(\sum_{i=1}^n w_i x_i\right) \leq \sum_{i=1}^n w_i f(x_i).
    \]
    Jika $f$ adalah fungsi konkaf, maka ketaksamaan di atas dibalik.
  \end{teorema}
  Biasanya bobot $w_i$ dinyatakan dalam bentuk $\frac{1}{n}$ untuk setiap $i$, sehingga ketaksamaan Jensen dapat ditulis sebagai
  \[
    f\left(\frac{x_1 + x_2 + \cdots + x_n}{n}\right) \leq \frac{f(x_1) + f(x_2) + \cdots + f(x_n)}{n}.
  \]
\end{frame}
