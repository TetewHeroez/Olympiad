\section{\textit{Lingkaran Dalam dan Lingkaran Luar}}
\begin{frame}{\insertsection}
  Pada topik ini kita ingin mengenali dua lingkaran penting pada segitiga:
  \begin{itemize}
    \item \textbf{Lingkaran dalam}, yang menyinggung ketiga sisi segitiga.
    \item \textbf{Lingkaran luar}, yang melalui ketiga titik sudut segitiga.
  \end{itemize}

  \vspace{0.4em}
  Dua pusat yang muncul pun berbeda:
  \begin{itemize}
    \item titik bagi sudut $\rightarrow$ \textbf{titik pusat lingkaran dalam} $I$;
    \item garis sumbu sisi $\rightarrow$ \textbf{titik pusat lingkaran luar} $O$.
  \end{itemize}
\end{frame}

\subsection{Lingkaran Dalam}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{definisi}[Lingkaran Dalam dan Titik Dalam]
        Pada segitiga $ABC$, \textbf{lingkaran dalam} adalah lingkaran yang menyinggung ketiga sisi segitiga. Pusatnya disebut \textbf{titik dalam} dan biasanya dinotasikan dengan $I$.
      \end{definisi}
      \begin{definisi}
        Titik dalam $I$ adalah titik yang berjarak sama terhadap ketiga sisi segitiga.
      \end{definisi}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95, every node/.style={font=\footnotesize}]
        \coordinate (A) at (0,3.2);
        \coordinate (B) at (-2.6,0);
        \coordinate (C) at (3,0);
        \coordinate (I) at (0.05,1.05);
        \coordinate (D) at ($(B)!(I)!(C)$);
        \coordinate (E) at ($(C)!(I)!(A)$);
        \coordinate (F) at ($(A)!(I)!(B)$);

        \draw[thick] (A)--(B)--(C)--cycle;
        \draw[blue, thick] (I) circle (1.05);
        \draw[dashed] (A)--(I);
        \draw[dashed] (B)--(I);
        \draw[dashed] (C)--(I);
        \draw[dashed] (I)--(D);

        \pic [draw, fill=green!20, angle radius=2.5mm] {right angle = I--D--C};

        \fill (A) circle (1.2pt) node[above] {$A$};
        \fill (B) circle (1.2pt) node[below left] {$B$};
        \fill (C) circle (1.2pt) node[below right] {$C$};
        \fill[red] (I) circle (1.6pt) node[right] {$I$};
        \fill (D) circle (1.1pt) node[below] {$D$};
        \fill (E) circle (1.1pt);
        \fill (F) circle (1.1pt);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Tiga garis bagi sudut dalam segitiga berpotongan di satu titik, yaitu titik dalam $I$.
  \end{teorema}

  \begin{teorema}
    Jika $r$ adalah jari-jari lingkaran dalam, $L$ luas segitiga, dan
    \[
      s = \frac{a+b+c}{2}
    \]
    adalah semikeliling segitiga, maka
    \[
      L = r s.
    \]
  \end{teorema}

  \begin{catatan}
    Rumus $L=rs$ sangat berguna saat panjang sisi segitiga diketahui, karena kita cukup mencari luas segitiga dan semikelilingnya untuk memperoleh jari-jari lingkaran dalam.
  \end{catatan}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Diketahui segitiga dengan panjang sisi $13$, $14$, dan $15$.
    Tentukan jari-jari lingkaran dalamnya.

    \vspace{0.6em}
    \textbf{Penyelesaian.}
    \[
      s=\frac{13+14+15}{2}=21.
    \]
    Dengan rumus Heron,
    \[
      L=\sqrt{21(21-13)(21-14)(21-15)}=\sqrt{21\cdot 8\cdot 7\cdot 6}=84.
    \]
    Maka
    \[
      r=\frac{L}{s}=\frac{84}{21}=4.
    \]
    Jadi jari-jari lingkaran dalam segitiga tersebut adalah
    \[
      \boxed{4}.
    \]
  \end{contoh}
\end{frame}

\subsection{Lingkaran Luar}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{definisi}[Lingkaran Luar dan Titik Sumbu]
        \textbf{Lingkaran luar} adalah lingkaran yang melalui ketiga titik sudut segitiga.
        Pusatnya disebut \textbf{titik sumbu} atau \textit{circumcenter}, dan biasanya dinotasikan dengan $O$.
      \end{definisi}
      \begin{definisi}
        Titik sumbu $O$ adalah titik yang berjarak sama ke ketiga titik sudut segitiga:
        \[
          OA = OB = OC.
        \]
      \end{definisi}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95, every node/.style={font=\footnotesize}]
        \coordinate (A) at (0,3.2);
        \coordinate (B) at (-2.2,0.2);
        \coordinate (C) at (3,0);
        \coordinate (O) at (0.42,0.92);
        \coordinate (M) at ($(B)!0.5!(C)$);
        \coordinate (N) at ($(A)!0.5!(B)$);

        \draw[thick] (A)--(B)--(C)--cycle;
        \draw[blue, thick] (O) circle (2.28);
        \draw[dashed] (O)--(A);
        \draw[dashed] (O)--(B);
        \draw[dashed] (O)--(C);
        \draw[dashed] ($(M)!-1.2!(O)$)--($(M)!1.2!(O)$);
        \draw[dashed] ($(N)!-1.15!(O)$)--($(N)!1.15!(O)$);

        \fill (A) circle (1.2pt) node[above] {$A$};
        \fill (B) circle (1.2pt) node[below left] {$B$};
        \fill (C) circle (1.2pt) node[below right] {$C$};
        \fill[red] (O) circle (1.6pt) node[right] {$O$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Tiga garis sumbu sisi segitiga berpotongan di satu titik, yaitu titik sumbu $O$.
  \end{teorema}

  \begin{teorema}
    Jika $R$ adalah jari-jari lingkaran luar segitiga $ABC$, maka berlaku
    \[
      \frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C} = 2R.
    \]
    Selain itu,
    \[
      R = \frac{abc}{4L},
    \]
    dengan $L$ adalah luas segitiga.
  \end{teorema}

  \begin{akibat}
    Pada segitiga siku-siku, titik sumbu terletak di titik tengah sisi miring. Karena itu jari-jari lingkaran luarnya sama dengan setengah panjang sisi miring.
  \end{akibat}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Segitiga siku-siku $ABC$ memiliki kaki-kaki $6$ dan $8$, sehingga sisi miringnya $10$.
    Tentukan jari-jari lingkaran luarnya.

    \vspace{0.6em}
    \textbf{Penyelesaian.}
    Karena segitiga siku-siku, pusat lingkaran luar terletak di titik tengah sisi miring.
    Maka
    \[
      R = \frac{10}{2}=5.
    \]
    Jadi jari-jari lingkaran luarnya adalah
    \[
      \boxed{5}.
    \]
  \end{contoh}
\end{frame}

\section{\textit{Segi Empat Tali Busur}}
\begin{frame}{\insertsection}
  Segi empat tali busur adalah salah satu objek geometri lingkaran yang paling sering muncul pada soal olimpiade.

  \vspace{0.5em}
  Ide yang paling sering dipakai:
  \begin{itemize}
    \item jumlah sudut berhadapan adalah $180^\circ$;
    \item sudut luar sama dengan sudut dalam berhadapan;
    \item kriteria kebalikannya juga sering dipakai untuk membuktikan empat titik sebidang lingkaran.
  \end{itemize}
\end{frame}

\subsection{Definisi dan Sifat Dasar}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{definisi}[Segi Empat Tali Busur]
        Segi empat $ABCD$ disebut \textbf{segi empat tali busur} atau \textbf{segi empat siklik} jika keempat titik $A,B,C,D$ terletak pada satu lingkaran yang sama.
      \end{definisi}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=1, every node/.style={font=\footnotesize}]
        \coordinate (O) at (0,0);
        \coordinate (A) at (110:2);
        \coordinate (B) at (20:2);
        \coordinate (C) at (-45:2);
        \coordinate (D) at (-150:2);

        \draw[blue, thick] (O) circle (2);
        \draw[thick] (A)--(B)--(C)--(D)--cycle;

        \fill (A) circle (1.2pt) node[above left] {$A$};
        \fill (B) circle (1.2pt) node[above right] {$B$};
        \fill (C) circle (1.2pt) node[below right] {$C$};
        \fill (D) circle (1.2pt) node[below left] {$D$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Sudut Berhadapan]
    Jika $ABCD$ adalah segi empat tali busur, maka
    \[
      \angle A + \angle C = 180^\circ
      \qquad \text{dan} \qquad
      \angle B + \angle D = 180^\circ.
    \]
  \end{teorema}

  \begin{teorema}[Kriteria Kebalikan]
    Jika pada segi empat $ABCD$ berlaku
    \[
      \angle A + \angle C = 180^\circ,
    \]
    maka keempat titik $A,B,C,D$ terletak pada satu lingkaran. Artinya, $ABCD$ adalah segi empat tali busur.
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Sudut Luar]
    Pada segi empat tali busur, sudut luar di suatu titik sama dengan sudut dalam yang berhadapan.
  \end{teorema}

  \vspace{0.3em}
  Misalnya, jika sisi $BC$ diperpanjang ke titik $E$, maka
  \[
    \angle ACE = \angle ABD
  \]
  dan hubungan-hubungan serupa lainnya juga berlaku.

  \begin{catatan}
    Dalam banyak soal, cara tercepat membuktikan suatu segi empat siklik adalah menunjukkan satu pasang sudut berhadapan saling suplementer atau satu sudut luar sama dengan sudut dalam berhadapan.
  \end{catatan}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Misalkan $ABCD$ adalah segi empat tali busur. Jika $\angle BAD = 72^\circ$, tentukan $\angle BCD$.

    \vspace{0.6em}
    \textbf{Penyelesaian.}
    Karena $ABCD$ segi empat tali busur, sudut-sudut yang berhadapan jumlahnya $180^\circ$. Maka
    \[
      \angle BAD + \angle BCD = 180^\circ.
    \]
    Jadi
    \[
      \angle BCD = 180^\circ - 72^\circ = 108^\circ.
    \]
    Dengan demikian,
    \[
      \boxed{\angle BCD = 108^\circ}.
    \]
  \end{contoh}
\end{frame}

\section{\textit{Titik Kuasa}}
\begin{frame}{\insertsection}
  Titik kuasa menghubungkan panjang-panjang ruas garis yang berkaitan dengan sebuah lingkaran.

  \vspace{0.5em}
  Tiga bentuk yang paling sering dipakai adalah:
  \begin{itemize}
    \item dua tali busur yang berpotongan di dalam lingkaran;
    \item dua secan dari satu titik luar;
    \item sebuah garis singgung dan sebuah secan dari titik luar yang sama.
  \end{itemize}
\end{frame}

\subsection{Definisi}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{definisi}[Titik Kuasa]
        Misalkan $P$ adalah sebuah titik dan $\omega$ adalah lingkaran.
        Jika suatu garis melalui $P$ memotong lingkaran di $A$ dan $B$, maka \textbf{kuasa titik $P$ terhadap $\omega$} didefinisikan sebagai
        \[
          \operatorname{Pow}_{\omega}(P)=PA \cdot PB.
        \]
      \end{definisi}

      Nilai ini tidak bergantung pada garis secan yang dipilih.
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95, every node/.style={font=\footnotesize}]
        \coordinate (O) at (0,0);
        \coordinate (P) at (3,1.7);
        \coordinate (A) at (18:2);
        \coordinate (B) at (155:2);

        \draw[blue, thick] (O) circle (2);
        \draw[thick] (P)--($(P)!1.2!(B)$);

        \fill[red] (P) circle (1.4pt) node[right] {$P$};
        \fill (A) circle (1.2pt) node[right] {$A$};
        \fill (B) circle (1.2pt) node[left] {$B$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Dua Tali Busur Berpotongan]
    Jika dua tali busur $AB$ dan $CD$ berpotongan di titik $P$ di dalam lingkaran, maka
    \[
      PA \cdot PB = PC \cdot PD.
    \]
  \end{teorema}

  \begin{teorema}[Dua Secan dari Titik Luar]
    Jika dari titik luar $P$ ditarik dua secan yang memotong lingkaran di
    $A,B$ dan $C,D$, maka
    \[
      PA \cdot PB = PC \cdot PD.
    \]
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Garis Singgung dan Secan]
    Jika dari titik luar $P$ ditarik garis singgung $PT$ dan secan yang memotong lingkaran di $A$ dan $B$, maka
    \[
      PT^2 = PA \cdot PB.
    \]
  \end{teorema}

  \begin{catatan}
    Dalam praktik, teorema titik kuasa hampir selalu dipakai untuk menukar informasi geometri lingkaran menjadi persamaan aljabar yang sederhana.
  \end{catatan}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Dari titik luar $P$ ditarik sebuah garis singgung $PT$ dan sebuah secan yang memotong lingkaran di $A$ dan $B$.
    Diketahui $PA=4$ dan $PB=9$. Tentukan panjang $PT$.

    \vspace{0.6em}
    \textbf{Penyelesaian.}
    Dengan teorema garis singgung-secan,
    \[
      PT^2 = PA \cdot PB = 4 \cdot 9 = 36.
    \]
    Karena panjang ruas garis bernilai positif, maka
    \[
      PT = \sqrt{36} = 6.
    \]
    Jadi,
    \[
      \boxed{PT = 6}.
    \]
  \end{contoh}
\end{frame}

\section*{\textit{Ringkasan}}
\begin{frame}{\insertsection}
  \begin{itemize}
    \item \textbf{Lingkaran dalam}: pusatnya titik potong garis bagi sudut, dan luas segitiga memenuhi $L=rs$.
    \item \textbf{Lingkaran luar}: pusatnya titik potong garis sumbu sisi, dan
          \[
            \frac{a}{\sin A}=\frac{b}{\sin B}=\frac{c}{\sin C}=2R.
          \]
    \item \textbf{Segi empat tali busur}: sudut-sudut berhadapan berjumlah $180^\circ$.
    \item \textbf{Titik kuasa}: produk panjang ruas pada konfigurasi secan/tangen bernilai sama.
  \end{itemize}

  \vspace{0.6em}
  Semua topik ini sangat sering muncul dalam \textit{angle chasing}, pembuktian siklik, dan pengolahan panjang ruas pada geometri olimpiade.
\end{frame}
