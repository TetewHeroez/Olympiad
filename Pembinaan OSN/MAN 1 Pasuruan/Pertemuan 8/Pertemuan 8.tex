\begin{frame}{\insertsection}{\insertsubsection}

  \begin{teorema}
    Jika $r$ adalah jari-jari lingkaran dalam, $L$ luas segitiga, dan
    $
      s = \frac{a+b+c}{2}
    $
    adalah semikeliling segitiga, maka
    $
      L = r s.
    $
  \end{teorema}

  \begin{teorema}
    Misalkan $I$ adalah titik pusat lingkaran dalam segitiga $ABC$ dan $r$ adalah jari-jari lingkaran dalamnya. Maka
    \[
      AI=\frac{r}{\sin\frac{A}{2}},\quad
      BI=\frac{r}{\sin\frac{B}{2}},\quad
      CI=\frac{r}{\sin\frac{C}{2}}.
    \]
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{contoh}
    Diketahui segitiga dengan panjang sisi $13$, $14$, dan $15$.
    Tentukan jari-jari lingkaran dalamnya.
    \vspace{0.6em}
    \textbf{Penyelesaian.}
    \[
      s=\frac{13+14+15}{2}=21.
    \]
    Dengan rumus Heron,
    \[
      L=\sqrt{21(21-13)(21-14)(21-15)}=\sqrt{21\cdot 8\cdot 7\cdot 6}=84.
    \]
    Maka
    $
      r=\frac{L}{s}=\frac{84}{21}=4.
    $
  \end{contoh}
\end{frame}

  \begin{teorema}[Hubungan Luas dan Semiperimeter]
    Misalkan $L$ adalah luas segitiga, $s$ adalah setengah keliling segitiga, dan $r_1, r_2, r_3$ berturut-turut adalah jari-jari lingkaran singgung luar yang menyinggung sisi $a, b, c$. Maka berlaku:
    \begin{align*}
      r_1  = \frac{L}{s - a}, \quad
      r_2  = \frac{L}{s - b}, \quad
      r_3  = \frac{L}{s - c}
    \end{align*}
  \end{teorema}

  \begin{lema}[Hubungan Jari-jari dengan Sudut]
    \begin{align*}
      r_1  = s \cdot \tan\frac{A}{2}, \quad
      r_2  = s \cdot \tan\frac{B}{2}, \quad
      r_3  = s \cdot \tan\frac{C}{2}
    \end{align*}
  \end{lema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Jarak Titik Sudut ke Excenter]
    Misalkan $I_1, I_2, I_3$ adalah pusat lingkaran singgung luar pada masing-masing sisi. Jarak dari titik sudut segitiga ke excenter yang bersesuaian adalah:
    \begin{align*}
      AI_1  = 4R \cdot \cos\frac{B}{2} \cdot \cos\frac{C}{2}, \quad
      BI_2  = 4R \cdot \cos\frac{C}{2} \cdot \cos\frac{A}{2}, \quad
      CI_3  = 4R \cdot \cos\frac{A}{2} \cdot \cos\frac{B}{2}
    \end{align*}
  \end{teorema}

  \begin{lema}[Jarak Incenter ke Excenter]
    Misalkan $I$ adalah pusat lingkaran dalam (Incenter) segitiga. Jarak dari $I$ ke masing-masing excenter adalah:
    \begin{align*}
      II_1  = 4R \sin\frac{A}{2}, \quad
      II_2  = 4R \sin\frac{B}{2}, \quad
      II_3  = 4R \sin\frac{C}{2}
    \end{align*}
  \end{lema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{proposisi}
    Misalkan $\omega$ adalah lingkaran berpusat di $O$, dan misalkan $T$ adalah titik pada $\omega$. Suatu garis $\ell$ yang melalui $T$ merupakan garis singgung lingkaran $\omega$ di titik $T$ jika dan hanya jika
    $
      OT \perp \ell.
    $
  \end{proposisi}
  \begin{teorema}[Alternate Segment Theorem]
    Misalkan $\omega$ adalah lingkaran dan garis $\ell$ menyinggung $\omega$ di titik $A$. Jika $B$ dan $C$ adalah titik-titik pada $\omega$, maka sudut antara garis singgung $\ell$ dan tali busur $AC$ sama dengan sudut keliling yang menghadap tali busur $AC$, yaitu
    \[
      \angle(\ell,AC)=\angle ABC.
    \]
  \end{teorema}
\end{frame}

\subsection{Titik Kuasa}

\begin{frame}{\insertsection}{\insertsubsection}

  \begin{definisi}[Titik Kuasa]
    Misalkan $P$ adalah sebuah titik dan $\omega$ adalah lingkaran.
    Jika $O$ adalah pusat lingkaran $\omega$ dan $r$ adalah jari-jarinya, maka \textbf{titik kuasa} $P$ terhadap $\omega$ didefinisikan sebagai
    \[
      \text{Pow}_{\omega}(P) = PO^2 - r^2 = (OP-r)(OP+r).
    \]
  \end{definisi}

  \begin{catatan}
    Tanda titik kuasa menunjukkan posisi relatif titik $P$ terhadap lingkaran $\omega$:
    \begin{itemize}
      \item Jika $\text{Pow}_{\omega}(P) > 0$, maka $P$ berada di luar lingkaran.
      \item Jika $\text{Pow}_{\omega}(P) = 0$, maka $P$ berada pada lingkaran.
      \item Jika $\text{Pow}_{\omega}(P) < 0$, maka $P$ berada di dalam lingkaran.
    \end{itemize}
  \end{catatan}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Dua Tali Busur Berpotongan \label{thm:power_of_point}]
    Jika dua tali busur $AB$ dan $CD$ (atau perpanjangannya) berpotongan di titik $P$ di dalam (atau di luar) lingkaran, maka
    \[
      PA \cdot PB = PC \cdot PD.
    \]
  \end{teorema}

  \begin{teorema}[Garis Singgung dan Secan \label{thm:tangent_secant}]
    Jika dari titik luar $P$ ditarik garis singgung $PT$ dan secan yang memotong lingkaran di $A$ dan $B$, maka
    \[
      PT^2 = PA \cdot PB.
    \]
  \end{teorema}

  \begin{teorema}[Dua Garis Singgung \label{thm:tangent_tangent}]
    Jika dari titik luar $P$ ditarik dua garis singgung yang menyinggung lingkaran di $T_1$ dan $T_2$, maka
    $
      PT_1 = PT_2.
    $
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}
  \begin{teorema}[Sudut Berhadapan]
    Jika $ABCD$ adalah segi empat tali busur, maka
    \[
      \angle A + \angle C = 180^\circ
      \quad \text{dan} \quad
      \angle ADB = \angle BCA
    \]
  \end{teorema}

  \begin{teorema}[Kriteria Segi Empat Tali Busur]
    Misal $ABCD$ merupakan sebuah segi empat. Ketiga pernyataan berikut ekuivalen:
    \begin{enumerate}
      \item $ABCD$ adalah segi empat tali busur.
      \item $\angle ABC + \angle CDA = 180^\circ$.
      \item $\angle ABD = \angle ACD$.
    \end{enumerate}
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}
  \begin{teorema}[Teorema Ptolemy]
    Hasil kali panjang kedua diagonal segiempat tali busur sama dengan jumlah hasil kali panjang sisi-sisi yang saling berhadapan.
    \begin{align*}
      p \cdot q & = a \cdot c + b \cdot d
    \end{align*}
  \end{teorema}

  \begin{teorema}[Teorema Kedua Ptolemy / Rasio Diagonal]
    Jika $p$ dan $q$ adalah panjang diagonal $AC$ dan $BD$ pada segi empat tali busur $ABCD$, maka
    \begin{align*}
      p & = \sqrt{\frac{(ac + bd)(ad + bc)}{ab + cd}} \\
      q & = \sqrt{\frac{(ac + bd)(ab + cd)}{ad + bc}}
    \end{align*}
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}
  \begin{teorema}[Formula Brahmagupta]
    Luas ($K$) dari sebuah segiempat tali busur dapat dihitung menggunakan panjang keempat sisinya dengan rumus:
    \begin{align*}
      K & = \sqrt{(s - a)(s - b)(s - c)(s - d)}
    \end{align*}
  \end{teorema}

  \begin{teorema}[Formula Jari-jari Luar Parameshvara]
    Jika $R$ adalah jari-jari lingkaran luar (circumradius) yang mengelilingi segiempat tali busur tersebut, maka panjang $R$ dapat dirumuskan sebagai:
    \begin{align*}
      R & = \frac{1}{4} \sqrt{\frac{(ab + cd)(ac + bd)(ad + bc)}{(s - a)(s - b)(s - c)(s - d)}}
    \end{align*}
  \end{teorema}
\end{frame}
