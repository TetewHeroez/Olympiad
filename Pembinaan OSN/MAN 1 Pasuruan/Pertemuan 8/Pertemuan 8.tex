\section{\textit{Angle Chasing}}
\begin{frame}{\insertsection}
  \textbf{\textit{Angle chasing}} adalah sebuah teknik dalam geometri yang digunakan untuk menentukan besar sudut-sudut dalam suatu gambar atau diagram dengan menggunakan hubungan antar sudut yang sudah diketahui.

  \begin{definisi}[Garis, Sinar, dan Segmen]
    \begin{enumerate}
      \item \textbf{Garis} adalah sebuah bentuk satu dimensi yang lurus dan \textbf{tidak mempunyai ujung}, atau tak terhingga.
      \item \textbf{Sinar} adalah sebuah bentuk satu dimensi yang lurus yang \textbf{dimulai dari satu pangkal} dan \textbf{memanjang tak hingga} dari titik tersebut.
      \item \textbf{Segmen} atau \textbf{Ruas Garis} adalah bagian dari garis yang dibatasi oleh dua titik di garis tersebut, sehingga panjangnya berhingga dan positif.
      \item Dua garis $\ell_1$ dan $\ell_2$ dikatakan \textbf{sejajar} apabila kedua garis \textbf{tidak berpotongan di titik manapun}. Dua sinar atau segmen dikatakan sejajar jika ketika keduanya diperpanjang menjadi garis, kedua garis tersebut sejajar.
    \end{enumerate}
  \end{definisi}
\end{frame}

\subsection{Sudut}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{proposisi}[Sudut Dasar\label{prop:sudut_dasar}]
    Sudut dinyatakan dalam satuan derajat, yang dilambangkan dengan tanda $^\circ$. Ukuran $360^\circ$ melambangkan satu putaran penuh.

    \vspace{0.5em}
    \begin{center}
      \begin{tikzpicture}
        \filldraw (0,0) circle (1pt) node[above] {$A$};
        \draw (0,0) -- (3,0);
        \filldraw (3,0) circle (1pt) node[above] {$B$};

        \draw[blue, thick, -stealth] (3.5,0) arc (0:360:0.5);
        \node[blue, right] at (3.5,0) {$360^\circ$};
      \end{tikzpicture}
    \end{center}
    \vspace{0.5em}

    Jika $C$ terletak pada garis $AB$, dan berada di antara $A$ dan $B$, maka $\angle ACB = 180^\circ$.

    \vspace{0.5em}
    \begin{center}
      \begin{tikzpicture}
        \filldraw (0,0) circle (1pt) node[left] {$A$};
        \draw (0,0) -- (4,0);
        \filldraw (4,0) circle (1pt) node[right] {$B$};

        \filldraw (2,0) circle (1pt) node[below] {$C$};

        \draw[blue, thick, stealth-stealth] (1.6,0) arc (180:0:0.4);
        \node[blue, above] at (2,0.4) {$180^\circ$};
      \end{tikzpicture}
    \end{center}
  \end{proposisi}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{proposisi}[Ekivalensi Sudut Transversal\label{prop:transversal_dua_garis_sejajar}]
    Misal garis $AB$ dan $CD$ sejajar, $P$ dan $Q$ adalah titik perpotongan garis transversal $XY$ dengan $AB$ dan $CD$ berturut-turut.
    \begin{center}
      \begin{tikzpicture}[scale=1, every node/.style={font=\footnotesize}]
        \def\ang{30}
        \def\radPQ{0.5} % Radius lingkaran P dan Q
        \def\radPt{1pt} % Radius titik (dot)

        \coordinate (C) at (0,0);
        \coordinate (D) at (6,0);
        \coordinate (A) at (0,1);
        \coordinate (B) at (6,1);

        \coordinate (Q) at (2.5,0);


        \path[name path=lineAB] (A) -- (B);
        \path[name path=lineQ] (Q) -- ++(\ang:4);
        \path [name intersections={of=lineAB and lineQ, by=P}];

        \coordinate (X) at ($(P) + (\ang:1.5)$);
        \coordinate (Y) at ($(Q) + ({180+\ang}:1.5)$);

          \begin{scope}
            \clip (P) circle (\radPQ);
            \fill[blue!30] (P) -- ++(0:1) arc (0:\ang:1) -- cycle;
            \fill[orange!40] (P) -- ++(\ang:1) arc (\ang:180:1) -- cycle;
            \fill[blue!30] (P) -- ++(180:1) arc (180:{180+\ang}:1) -- cycle;
            \fill[orange!40] (P) -- ++({180+\ang}:1) arc ({180+\ang}:360:1) -- cycle;
          \end{scope}
          \draw (P) circle (\radPQ);

          \begin{scope}
            \clip (Q) circle (\radPQ);
            \fill[blue!30] (Q) -- ++(0:1) arc (0:\ang:1) -- cycle;
            \fill[orange!40] (Q) -- ++(\ang:1) arc (\ang:180:1) -- cycle;
            \fill[blue!30] (Q) -- ++(180:1) arc (180:{180+\ang}:1) -- cycle;
            \fill[orange!40] (Q) -- ++({180+\ang}:1) arc ({180+\ang}:360:1) -- cycle;
          \end{scope}
          \draw (Q) circle (\radPQ);

          \draw (A) -- (B);
          \draw (C) -- (D);
          \draw (Y) -- (X);

          \filldraw (A) circle (\radPt) node[left] {$A$};
          \filldraw (B) circle (\radPt) node[right] {$B$};
          \filldraw (C) circle (\radPt) node[left] {$C$};
          \filldraw (D) circle (\radPt) node[right] {$D$};
          \filldraw (X) circle (\radPt) node[right] {$X$};
          \filldraw (Y) circle (\radPt) node[left] {$Y$};
          \filldraw (P) circle (\radPt);
          \filldraw (Q) circle (\radPt);

          \node at ($(P) + ({90+(\ang/2)}:{0.25})$) {$P$};
          \node at ($(Q) + ({90+(\ang/2)}:{0.25})$) {$Q$};

      \end{tikzpicture}
    \end{center}
    Maka, berlaku persamaan sudut sebagai berikut:
    \[
      \angle APX = \angle BPY = \angle CQX = \angle DQY \quad \text{dan} \quad \angle BPX = \angle APY = \angle DQX = \angle CQY.
    \]
  \end{proposisi}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  Beberapa sebutan lain terkait hubungan antar sudut pada \myref{prop:transversal_dua_garis_sejajar}:
  \begin{itemize}
    \item \textbf{Sudut Sehadap}
          \[\angle APX = \angle CQX, \quad \angle BPY = \angle DQY, \quad \angle BPX = \angle DQX, \quad \angle APY = \angle CQY\]
    \item \textbf{Sudut Dalam Berseberangan}
          \[\angle BPY = \angle CQX, \quad \angle APY = \angle DQX\]
    \item \textbf{Sudut Luar Berseberangan}
          \[\angle APX = \angle DQY, \quad \angle BPX = \angle CQY\]
    \item \textbf{Sudut Bertolak Belakang}
          \[\angle APX = \angle BPY, \quad \angle CQX = \angle DQY, \quad \angle BPX = \angle APY, \quad \angle DQX = \angle CQY\]
  \end{itemize}
\end{frame}

\subsection{Sudut Poligon}
\begin{frame}{\insertsection}{\insertsubsection}
  \begin{definisi}[Sudut Interior dan Eksterior Poligon]
    Diberikan sebuah poligon tertutup dengan $n$ sisi. Pada setiap titik sudut $p_k$ (untuk $k = 1, 2, \dots, n$), didefinisikan:
    \begin{itemize}
      \item \textbf{Sudut Interior} ($\alpha_k$): Sudut yang dibentuk oleh dua sisi poligon yang saling bertemu di titik $p_k$, dan diukur pada bagian dalam area poligon tertutup tersebut.
      \item \textbf{Sudut Eksterior} ($\beta_k$): Sudut yang dibentuk oleh salah satu sisi poligon dengan perpanjangan garis lurus dari sisi yang bersebelahan dengannya di titik $p_k$.
    \end{itemize}
    Pada setiap titik sudut poligon, pasangan sudut interior dan eksterior ini saling berpelurus, sehingga selalu memenuhi persamaan $\alpha_k + \beta_k = 180^\circ$
  \end{definisi}
  \begin{alertblock}{Catatan}
    Pada poligon cekung (non-konveks), sudut eksterior pada titik sudut yang merefleks ke dalam dapat bernilai negatif agar masih memenuhi hubungan $\alpha_k + \beta_k = 180^\circ$.
  \end{alertblock}
\end{frame}

  \begin{definisi}[Lingkaran Dalam / \textit{Incircle}]

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}[Perbandingan Garis Bagi\label{thm:perbandingan_garis_bagi}]
    Pada $\triangle ABC$, jika $AD$ adalah garis bagi sudut $A$ yang ditarik dari titik $A$ ke sisi $BC$, maka berlaku
    \[
      \frac{\overline{BD}}{\overline{DC}}=\frac{\overline{AB}}{\overline{AC}}.
    \]
    Selain itu, berlaku pula
    \[
      \frac{\overline{BD}}{\overline{BC}}=\frac{\overline{AB}}{\overline{AB}+\overline{AC}},
      \qquad
      \frac{\overline{DC}}{\overline{BC}}=\frac{\overline{AC}}{\overline{AB}+\overline{AC}}.
    \]
  \end{teorema}
  \begin{teorema}[Jari-jari Lingkaran Dalam\label{thm:jari_jari_lingkaran_dalam}]
    Misalkan $r$ adalah jari-jari lingkaran dalam pada $\triangle ABC$, $L$ adalah luas $\triangle ABC$, dan $s$ adalah setengah keliling segitiga yaitu $s = \frac{a+b+c}{2}$. Maka berlaku hubungan
    $ r = \frac{L}{s} $
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Pada $\triangle ABC$, jika $AD$ adalah garis bagi sudut $A$ yang memotong sisi $BC$ di titik $D$, maka panjang garis bagi $AD$ dapat dihitung dengan:
    \[
      \overline{AD}^2 = \overline{AB} \cdot \overline{AC} - \overline{BD} \cdot \overline{DC}, \quad \text{atau} \quad \overline{AD}^2 = \overline{AB} \cdot \overline{AC} \left(1 - \frac{\overline{BC}^2}{(\overline{AB} + \overline{AC})^2}\right).
    \]
    Atau jika menggunakan sudut, panjang $\overline{AD}$ dapat dicari dengan:
    \[
      \overline{AD}=\frac{2\cdot \overline{AB}\cdot \overline{AC}\cdot \cos\left(\frac{\angle A}{2}\right)}{\overline{AB}+\overline{AC}}.
    \]
  \end{teorema}
\end{frame}

  \end{definisi}

    Lingkaran yang kelilingnya melalui ketiga titik sudut sebuah segitiga. Pusat dari lingkaran ini adalah titik sumbu.
  \end{definisi}

  \begin{lema}
    Pada sebuah segitiga, hanya ada satu (tunggal) titik sumbu yang merupakan perpotongan dari ketiga garis sumbu sisi-sisi segitiga tersebut.
  \end{lema}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{teorema}
    Jarak dari titik sumbu $O$ ke masing-masing titik sudut segitiga ($A$, $B$, dan $C$) adalah sama panjang dan merupakan jari-jari ($R$) dari lingkaran luar segitiga. Jika panjang sisi-sisi segitiga adalah $a$, $b$, dan $c$, serta luas segitiga adalah $L$, maka panjang jari-jari lingkaran luar memenuhi persamaan:
    \[
      R = \frac{a \cdot b \cdot c}{4L}
    \]
  \end{teorema}

  \begin{teorema}
    Jari-jari lingkaran luar $R$ juga memiliki hubungan langsung dengan Hukum Sinus pada trigonometri segitiga, yaitu:
    \[
      \frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C} = 2R
    \]
  \end{teorema}

\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{definisi}[Titik Tinggi / \textit{Orthocenter}]
    Titik perpotongan tunggal dari ketiga garis tinggi (atau perpanjangannya) di dalam sebuah segitiga.
  \end{definisi}
  \begin{teorema}
    Jika panjang suatu sisi alas adalah $a$ dan panjang garis tinggi yang bersesuaian dengan alas tersebut adalah $t_a$, maka luas segitiga ($L$) adalah:
    \[
      L = \frac{1}{2} \cdot a \cdot t_a
    \]
  \end{teorema}
\end{frame}

\begin{frame}{\insertsection}{\insertsubsection}
  \begin{lema}
    Jika suatu median pada segitiga juga merupakan garis tinggi,
    maka segitiga tersebut sama kaki.
  \end{lema}
  \begin{akibat}
    Pada sebarang segitiga, jika $t_a$, $t_b$, dan $t_c$ masing-masing adalah panjang garis tinggi yang bersesuaian dengan sisi $a$, $b$, dan $c$, maka berlaku hubungan:
    \[
      a \cdot t_a = b \cdot t_b = c \cdot t_c = 2L
    \]
  \end{akibat}
\end{frame}
